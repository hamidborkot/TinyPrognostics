%% --- Abstract ---
\abstract{
Prognostics and health management (PHM) systems deployed on edge hardware face
a fundamental conflict: state-of-the-art deep learning models for remaining
useful life (RUL) estimation and fault classification demand computational
resources that microcontroller-class devices cannot provide. We introduce
NanoSentry, a 12{,}052-parameter (47.1\,KB) unified neural architecture that
resolves this conflict by jointly performing RUL regression and health-state
classification across heterogeneous industrial domains within a single model.
NanoSentry combines dilated causal 1D convolutions with a novel CrossSensorGate
mechanism for adaptive inter-sensor feature weighting and a GRU with orthogonal
initialisation for temporal context. Evaluated on three public benchmarks ---
NASA C-MAPSS turbofan engine degradation (four subsets), NASA lithium-ion
battery state-of-health, and CWRU rolling bearing fault classification ---
NanoSentry achieves RMSE of 14.75 on C-MAPSS FD001, surpassing an LSTM
baseline 4.5$\times$ its size, and 99.96\% fault classification accuracy.
Critically, in the operationally decisive window of RUL $\leq$ 30 cycles,
NanoSentry reduces prediction error by 3--7$\times$ compared with a
comparable-size CNN baseline whose critical-zone RMSE pathologically
exceeds its overall error --- a hazardous property for safety-critical
maintenance. To our knowledge, NanoSentry is the first sub-50\,KB architecture
validated simultaneously across turbofan, battery, and bearing prognostic tasks.
}

\subsection{Contributions}\label{sec:intro:contributions}
The main contributions of this work are as follows:
\begin{enumerate}
 \item We propose NanoSentry, a 12{,}052-parameter unified PHM architecture
 that jointly performs RUL regression and health-state classification across
 three industrial domains within a single 47.1\,KB model.
 \item We introduce the CrossSensorGate mechanism, a lightweight attention
 module that adaptively weights inter-sensor features without increasing
 model depth or parameter count.
 \item We demonstrate, for the first time, that a sub-50\,KB architecture
 achieves competitive accuracy against models 4.5$\times$ its size on the
 NASA C-MAPSS benchmark.
 \item We establish a critical-zone RMSE evaluation protocol (RUL $\leq$ 30
 cycles) and show that NanoSentry reduces end-of-life prediction error by
 3--7$\times$ compared with a comparable-size CNN baseline.
\end{enumerate}

\subsection{Dilated Causal Convolution Block}\label{sec:arch:dilconv}
% TODO: Describe the TCN block.
Let $\mathbf{X} \in \mathbb{R}^{T \times C}$ denote an input window of
$T$ timesteps and $C$ sensor channels. A dilated causal convolution with
dilation factor $d$ and kernel size $k$ computes
\begin{equation}\label{eq:dilconv}
 \mathbf{h}_t = \text{ReLU}\!\left(
 \sum_{j=0}^{k-1} \mathbf{W}_j \, \mathbf{x}_{t - d \cdot j}
 + \mathbf{b} \right),
\end{equation}
where $\mathbf{W}_j$ are learnable filter weights and causality is enforced
by restricting attention to $t - d\cdot j \leq t$. Three successive blocks
with $d \in \{1,2,4\}$ yield a receptive field of $7(k-1)+1$ timesteps
without pooling.

\subsection{CrossSensorGate}\label{sec:arch:gate}
% TODO: Describe and motivate the gate.
Let $\mathbf{H} \in \mathbb{R}^{T \times d_{\text{model}}}$ be the TCN
output. The CrossSensorGate computes a channel-wise attention mask
\begin{equation}\label{eq:gate}
 \mathbf{g} = \sigma\!\left( \mathbf{W}_g \mathbf{H}^\top + \mathbf{b}_g \right),
 \qquad
 \hat{\mathbf{H}} = \mathbf{g}^\top \odot \mathbf{H},
\end{equation}
where $\sigma$ is the sigmoid function, $\odot$ denotes element-wise
multiplication, and $\mathbf{W}_g \in \mathbb{R}^{d_{\text{model}}
\times d_{\text{model}}}$. The gate suppresses sensor channels
irrelevant to the current degradation regime without adding layers.

\subsection{Residual Skip Connection}\label{sec:arch:skip}
A linear projection $\mathbf{W}_{\text{skip}} \in
\mathbb{R}^{C \times d_{\text{model}}}$ maps the raw input directly to
the model dimension, and the gated feature map is augmented by
\begin{equation}\label{eq:skip}
 \tilde{\mathbf{H}} = \hat{\mathbf{H}} + \mathbf{X}\mathbf{W}_{\text{skip}}.
\end{equation}
This preserves raw sensor gradients during backpropagation and
prevents vanishing-gradient degradation in shallow stacks.

\subsection{Gated Recurrent Unit}\label{sec:arch:gru}
The skip-augmented representation is passed to a single-layer GRU
\begin{equation}\label{eq:gru}
 \mathbf{z}_t = \text{GRU}\!\left(\tilde{\mathbf{h}}_t,\, \mathbf{z}_{t-1}\right),
\end{equation}
with hidden dimension $d = 24$ and weights initialised with an
orthogonal matrix~\cite{TODO} to prevent gradient explosion at the
start of training.

\subsection{Dual Output Heads}\label{sec:arch:heads}
The final hidden state $\mathbf{z}_T$ is fed to two independent
fully-connected heads:
\begin{equation}\label{eq:heads}
 \hat{y}_{\text{RUL}} = \mathbf{w}_r^\top \mathbf{z}_T + b_r,
 \qquad
 \hat{\mathbf{y}}_{\text{state}} = \text{softmax}\!\left(
 \mathbf{W}_c \mathbf{z}_T + \mathbf{b}_c \right).
\end{equation}

\subsection{Joint Loss Function}\label{sec:arch:loss}
The model is trained end-to-end by minimising
\begin{equation}\label{eq:loss}
 \mathcal{L} = \lambda_1 \, \mathcal{L}_{\text{RUL}}
 + \lambda_2 \, \mathcal{L}_{\text{state}},
\end{equation}
where $\mathcal{L}_{\text{RUL}}$ is mean squared error,
$\mathcal{L}_{\text{state}}$ is cross-entropy, and
$\lambda_1, \lambda_2$ are task weights set to 1.0 and 0.5 respectively.
The asymmetric NASA scoring function
\begin{equation}\label{eq:nasa}
 s_i =
\begin{cases}
 e^{-d_i / 13} - 1 & d_i < 0 \quad \text{(early prediction)}\\
 e^{\phantom{-}d_i / 10} - 1 & d_i \geq 0 \quad \text{(late prediction)}
\end{cases}
\end{equation}
is reported as a secondary metric to penalise late predictions more
severely, where $d_i = \hat{y}_i - y_i$.

\subsection{Training Procedure}\label{sec:arch:training}
\begin{algorithm}[!h]
\caption{NanoSentry training procedure}\label{alg:training}
\begin{algorithmic}[1]
\Require Dataset $\mathcal{D}$, epochs $E$, batch size $B$,
 learning rate $\eta$, task weights $\lambda_1, \lambda_2$
\Ensure Trained model parameters $\boldsymbol{\theta}$
\State Initialise $\boldsymbol{\theta}$ (GRU weights: orthogonal)
\State Standardise inputs per sensor channel
\For{epoch $e = 1, \ldots, E$}
 \For{each mini-batch $(\mathbf{X}, y_{\text{RUL}}, y_{\text{state}}) \in \mathcal{D}$}
 \State Forward pass: compute $\hat{y}_{\text{RUL}}$,
 $\hat{\mathbf{y}}_{\text{state}}$ via Eqs.\ (\ref{eq:dilconv})---(\ref{eq:heads})
 \State Compute $\mathcal{L}$ via Eq.\ (\ref{eq:loss})
 \State Backpropagate; update $\boldsymbol{\theta}$ with Adam
 \EndFor
 \State Decay $\eta$ with cosine schedule
\EndFor
\State \Return $\boldsymbol{\theta}$
\end{algorithmic}
\end{algorithm}

\subsection{C-MAPSS RUL Estimation}
\label{sec:results:cmapss}

Table~\ref{tab:cmapss} shows that NanoSentry achieves the lowest RMSE
on three of the four C-MAPSS subsets, outperforming the 210.3\,KB
LSTM-64 on FD001 (14.75 vs.\ 16.28) and FD003 (13.95 vs.\ 16.02)
despite being $4.5{\times}$ smaller, confirming that the CrossSensorGate
and GRU encoder recover representational capacity that a plain LSTM
requires far more parameters to achieve.
On the harder multi-condition subset FD002, NanoSentry also leads
(26.74 vs.\ 27.95), suggesting that the channel-attention mechanism
is particularly effective when irrelevant sensor readings must be
suppressed across heterogeneous operating regimes.
The sole exception is FD004, where LSTM-64 holds a narrow margin
(27.15 vs.\ 27.67, $\Delta{=}0.52$ cycles), indicating that the
additional recurrent capacity yields only a marginal benefit on the
most complex subset combining six operating conditions with two
concurrent fault modes.
Across all subsets, both NanoSentry and LSTM-64 reduce Ridge
Regression RMSE by 60--70\%, confirming the necessity of learned
temporal feature extraction over hand-crafted statistics.
The NASA asymmetric scores show that NanoSentry's scores on the
single-condition subsets FD001 (350.47) and FD003 (315.86) are
substantially lower than on the multi-condition subsets FD002
(7{,}405.94) and FD004 (8{,}677.78), reflecting the exponential
penalty structure of Eq.~\ref{eq:nasa}: the larger prediction variance
induced by six heterogeneous operating conditions amplifies
late-prediction errors multiplicatively, driving scores upward even
when RMSE differences are modest.
NASA Score values for CNN-32 and Ridge Regression were not computed
in the baseline evaluation, as those models were assessed primarily
on RMSE and critical-zone accuracy where their limitations are most
clearly exposed.

\subsection{Critical-Zone Analysis (RUL $\leq$ 30 Cycles)}
\label{sec:results:critical}

End-of-life prediction accuracy is the operationally critical metric
because maintenance decisions made within the final 30 cycles
directly determine whether a failure is intercepted or missed; a model
with acceptable global RMSE but inflated critical-zone error is unsafe
in practice regardless of its headline number.
The Crit.\ column of Table~\ref{tab:cmapss} reveals a pathological
failure of CNN-32: its critical-zone RMSE of 29.05 on FD001
\emph{exceeds} its global RMSE of 26.47, meaning the model is
systematically less accurate precisely when accuracy matters most,
a consequence of the CNN's fixed receptive field failing to track the
accelerating degradation slope near end-of-life
(Figure~\ref{fig:critical}).
NanoSentry reduces the FD001 critical-zone error to 4.63 cycles, a
$6.3{\times}$ improvement over CNN-32 and a $1.1{\times}$ improvement
over the larger LSTM-64 (5.27), demonstrating that the dilated
causal convolution stack combined with the GRU temporal encoder
captures the sharp late-life degradation dynamics that
convolutional-only architectures miss.
LSTM-64 leads critical-zone RMSE on FD002 (4.42 vs.\ 5.52) and FD003
(2.66 vs.\ 3.14), indicating that pure recurrent memory is
advantageous for multi-condition end-of-life tracking; addressing this
gap via domain-adaptive fine-tuning is left for future work.

\subsection{Multi-Domain Results and Transfer Learning}
\label{sec:results:multidomain}

Table~\ref{tab:multidomain} demonstrates that a single 47.1\,KB
NanoSentry model generalises across three physically distinct asset
classes without any domain-specific architectural modification.
On the NASA Battery dataset, NanoSentry achieves RMSE of 1.87 cycles
SOH-equivalent and MAE of 0.42 cycles when trained from scratch,
competitive with dedicated battery SOH methods \cite{Gong2024} trained
exclusively on battery data.
On the CWRU bearing fault diagnosis task, NanoSentry achieves
99.96\% classification accuracy across all ten health states and four
load conditions, matching CNN architectures specifically designed for
vibration signal classification \cite{Wen2018,Zhang2018} while
operating within the same sub-50\,KB footprint used for turbofan
RUL regression.
Together these results constitute, to the best of our knowledge, the
first empirical demonstration that a unified architecture below
50\,KB achieves competitive accuracy simultaneously across turbofan,
battery, and bearing PHM domains.

The lower half of Table~\ref{tab:multidomain} presents a notable
negative result: fine-tuning a NanoSentry model pre-trained on FD001
does not consistently outperform training from scratch on the target
domain at any data fraction tested.
On FD003, the scratch baseline (13.95) outperforms all frozen-backbone
transfer variants across every fraction (10\%: 15.08, 25\%: 15.35,
50\%: 14.64, 100\%: 15.31).
On the NASA Battery task, the gap is larger: scratch achieves 1.87
while full fine-tuning at 100\% data yields 3.75, confirming that
the domain shift between turbofan thermal degradation and
electrochemical capacity fade is sufficiently large to make
positive transfer difficult for a model with no domain-disentangled
latent space \cite{Almeida2026}.
We attribute this behaviour to the compactness of NanoSentry's
12{,}052-parameter encoder: with so few parameters, the model
converges rapidly from random initialisation even on small target
datasets, offering little gradient-step advantage to a pre-trained
starting point.
This finding motivates a federated PHM paradigm as a promising
future direction: rather than sequential fine-tuning, coordinating
gradient updates across heterogeneous asset-type clients while
keeping raw data local could allow NanoSentry to learn shared
degradation priors without suffering the domain-shift penalty
observed here \cite{Almeida2026}.

\subsection{Ablation Study}
\label{sec:results:ablation}

Table~\ref{tab:ablation} decomposes the contribution of each
NanoSentry component by removing one element at a time while keeping
all other hyperparameters fixed and retraining from scratch.
Removing the CrossSensorGate is the single most damaging
intervention, raising FD001 RMSE by 2.56 ($14.75 \to 17.31$) and
FD003 RMSE by 2.63 ($13.95 \to 16.58$), confirming that
channel-wise attention is not merely a regulariser but provides
functionally distinct capacity to suppress irrelevant sensor
readings that the GRU alone cannot recover.
Removing the residual skip connection causes the second-largest
degradation ($+$1.19 on FD001, $+$0.92 on FD003), demonstrating
that direct access to raw sensor values helps the GRU anchor its
hidden-state transitions to absolute degradation levels.
Fixing all dilation factors to $d{=}1$ (removing multi-scale
temporal coverage) produces a more modest increase ($+$1.53 on
FD001, $+$1.46 on FD003), indicating that while broader
receptive fields help, the CrossSensorGate and skip connection
together account for the majority of NanoSentry's advantage over
a plain single-scale TCN.

\subsection{Transfer Learning and Domain Specificity}\label{sec:disc:transfer}
The negative transfer result in Section~\ref{sec:results:multidomain}
reflects a fundamental property of physical degradation: turbofan
thermal creep, lithium-ion capacity fade, and bearing surface spalling
are governed by entirely different physicochemical mechanisms, so latent
features learned on one domain are not merely shifted but structurally
misaligned with those of another, making sequential fine-tuning an
unreliable strategy~\cite{Jiang2026}.
A more principled solution is federated PHM: by coordinating gradient
updates across multiple asset-type clients while keeping raw sensor
data local, it is possible to train a shared NanoSentry encoder that
learns domain-invariant degradation priors without any single client
exposing proprietary data~\cite{Almeida2026}.
